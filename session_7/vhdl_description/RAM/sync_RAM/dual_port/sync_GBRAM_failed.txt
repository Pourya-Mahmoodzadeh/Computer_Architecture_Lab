-- GBRAM = synchronous gate basesd random access memory
library ieee;
use ieee.std_logic_1164.all;
use ieee.numeric_std.all;

entity sync_GBRAM is
    generic (
        ADDR_WIDTH : integer := 10; 
        DATA_WIDTH : integer := 32
    );
    port (
        clk : in  std_logic;
        write : in  std_logic;
        read : in std_logic; -- we are asked to take a read signal as input
        waddr : in  std_logic_vector(ADDR_WIDTH - 1 downto 0); -- address to be wrtten to
        raddr : in std_logic_vector(ADDR_WIDTH - 1 downto 0); -- address to be read
        wdata : in  std_logic_vector(DATA_WIDTH - 1 downto 0); -- data to write
        rdata : out std_logic_vector(DATA_WIDTH - 1 downto 0); -- the data that has been read
        reset_n : in std_logic; -- asynchronous active-low reset
        -- conflict : out std_logic; -- when simultaneous read and write are happening using same address, this reliably writes, but reading output is not reliable, warned by this signal.
        data_is_written_reliably : out std_logic;
        data_is_read_reliably : out std_logic;
        acknowledge_read : in std_logic;
        acknowledge_write : in std_logic
    );
end entity;

architecture rtl of sync_GBRAM is

    type ram_type is array (0 to (2**ADDR_WIDTH) - 1) of std_logic_vector(DATA_WIDTH-1 downto 0);
    signal ram : ram_type;
    signal ram_flat : std_logic_vector(DATA_WIDTH * (2**ADDR_WIDTH) - 1 downto 0);
    signal mux_out : std_logic_vector(DATA_WIDTH - 1 downto 0);

    signal wplace : std_logic_vector(2**ADDR_WIDTH -1 downto 0);

begin

    CONNECT_RAM_FLAT: for i in 2**ADDR_WIDTH - 1 downto 0 generate
        ram_flat(DATA_WIDTH * (i + 1) - 1 downto DATA_WIDTH * i) <= ram(i);
    end generate;

    WRITE_LOGIC: for i in 0 to (2**ADDR_WIDTH - 1) generate
        process(clk)
        begin
            if rising_edge(clk) then
                if reset_n = '1' and write = '1' and wplace(i) = '1' then
                    ram(i) <= wdata;
                    data_is_written_reliably <= '1';
                end if;
            end if;
        end process;
    end generate;

    WDEC: entity work.decoder_rec 
        generic map(
            width => ADDR_WIDTH
        )
        port map(
            input => waddr,
            output => wplace,
            enable => '1'
        );


    RMUX: entity work.mux 
            generic map(
                SEL_WIDTH => ADDR_WIDTH,
                DATA_WIDTH => DATA_WIDTH
            )
            port map ( 
                sel => raddr,
                data => ram_flat,
                output => mux_out,
                enable => '1'
            );

    


    READ_LOGIC: process(clk)
    begin
        if rising_edge(clk) then
            if reset_n = '0' then 
                INITIALIZE : for i in 2**ADDR_WIDTH - 1 downto 0 loop 
                    ram(i) <= std_logic_vector(to_unsigned(i, DATA_WIDTH));
                    data_is_read_reliably <= '0';
                    data_is_written_reliably <= '0';
                    rdata <= ram(0);
                end loop;
            else 
                if (read = '1') and (write = '0') then
                    rdata <= mux_out;
                    data_is_read_reliably <= '1';
                end if;
            end if;
        end if;
    end process;

    ACKNOWLEDGEMENT: process (clk) 
    begin 
        if rising_edge(clk) then
            if reset_n = '1' and acknowledge_read = '1' then
                data_is_read_reliably <= '0';
            end if;
            if reset_n = '1' and acknowledge_write = '1' then
                data_is_written_reliably <= '0';
            end if;
        end if;
    end process;






end architecture;
